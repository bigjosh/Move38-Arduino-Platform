blink.name=Blink

blink.upload.tool=avrdude

# Arduino IDE 2.x (and arduino-cli) only pass the Tools->Programmer selection for
# Sketch->Upload Using Programmer. The plain Upload button needs a protocol of its own,
# otherwise it fails with "A programmer is required to upload" (issue #109).
# This menu supplies it. Upload Using Programmer still uses Tools->Programmer as before.
blink.upload.tool.default=avrdude
blink.menu.uploadvia.usbtinyisp=USBtinyISP
blink.menu.uploadvia.usbtinyisp.upload.protocol=usbtiny
blink.menu.uploadvia.usbtinyisp.upload.extra_params=
blink.menu.uploadvia.avrispmkii=AVRISP mkII
blink.menu.uploadvia.avrispmkii.upload.protocol=stk500v2
blink.menu.uploadvia.avrispmkii.upload.extra_params=-Pusb
blink.menu.uploadvia.usbasp=USBasp
blink.menu.uploadvia.usbasp.upload.protocol=usbasp
blink.menu.uploadvia.usbasp.upload.extra_params=-Pusb
blink.menu.uploadvia.avrisp=AVR ISP (serial port)
blink.menu.uploadvia.avrisp.upload.protocol=stk500v1
blink.menu.uploadvia.avrisp.upload.extra_params=-P{serial.port}
blink.menu.uploadvia.arduinoasisp=Arduino as ISP (serial port)
blink.menu.uploadvia.arduinoasisp.upload.protocol=arduino
blink.menu.uploadvia.arduinoasisp.upload.extra_params=-P{serial.port} -b19200
blink.menu.uploadvia.arduinoisp=ArduinoISP
blink.menu.uploadvia.arduinoisp.upload.protocol=arduinoisp
blink.menu.uploadvia.arduinoisp.upload.extra_params=

# This is the maximum size of a program image. It comes from 
# Used for printing the "Sketch uses... of program storage space..." message and for moving the program image from base 0x000 "active" game area up to the "built-in" game area. 
# Obnoxiously, the `obj-size` command seems to only be able to take decimal, so 5888=0x1700
# Must match the `FLASH_GAME_IMAGE_SIZE` in the linker file. 
blink.upload.maximum_size=5888
blink.upload.maximum_data_size=1024
blink.upload.speed=115200

# The only fuse that is different from factory default is extended_fuses. 
# BOOTRST=0 (programmed)  enables jumping to the bootloader on reset and sets the bootloader 
# BOOTSZ1=0, BOOTSZ0=0 (both programmed) sets bootloader size to 1Kb and is the default. 

blink.bootloader.tool=avrdude
blink.bootloader.low_fuses=0x62
blink.bootloader.high_fuses=0xDF
blink.bootloader.extended_fuses=0xf8

# Note that the folder `bootloaders` seems to be hardcoded into the Arduino IDE

blink.bootloader.file=BlinkBIOS-168PB.hex

blink.build.mcu=atmega168pb
blink.build.f_cpu=8000000L
blink.build.board=AVR_BLINK_168
blink.build.core=blinklib
blink.build.variant=standard
blink.build.linkscript=avr5-atmega168pb.xn
blink.build.builtinbase=0x1700
